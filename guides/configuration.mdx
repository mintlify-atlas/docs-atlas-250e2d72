A UDA project is configured through a set of XML and properties files generated by the project wizard. This page explains what each file does and where to make common customisations.

### applicationContext.xml

This file is the root Spring application context. It declares:

- `<context:component-scan>` — scans your base package for `@Service`, `@Repository`, and other annotated beans.
- `<tx:annotation-driven>` — enables `@Transactional` support.
- The `JpaTransactionManager` and `EntityManagerFactory` beans.

```xml
<!-- applicationContext.xml (excerpt) -->
<context:component-scan base-package="com.example.myapp" />

<tx:annotation-driven transaction-manager="transactionManager" />

<bean id="transactionManager"
      class="org.springframework.orm.jpa.JpaTransactionManager">
    <property name="entityManagerFactory" ref="entityManagerFactory" />
</bean>
```

<Tip>
  Add application-wide beans (e.g. caches, email senders, scheduled tasks) to this file rather than to `dispatcher-servlet.xml`, which is scoped to the web layer only.
</Tip>

### applicationContext-security.xml

Configures Spring Security. Key sections to customise:

```xml
<!-- applicationContext-security.xml (excerpt) -->
<security:http auto-config="true" use-expressions="true">

    <!-- Public resources -->
    <security:intercept-url pattern="/resources/**" access="permitAll" />
    <security:intercept-url pattern="/login"         access="permitAll" />

    <!-- Protected area — requires authentication -->
    <security:intercept-url pattern="/**" access="isAuthenticated()" />

    <security:form-login login-page="/login"
                         default-target-url="/home"
                         authentication-failure-url="/login?error" />
    <security:logout logout-success-url="/login?logout" />
</security:http>

<security:authentication-manager>
    <security:authentication-provider>
        <security:user-service>
            <security:user name="admin"
                           password="{noop}admin"
                           authorities="ROLE_ADMIN" />
        </security:user-service>
    </security:authentication-provider>
</security:authentication-manager>
```

Common customisation points:

| Task | What to change |
|---|---|
| Add a new protected URL pattern | Add `<security:intercept-url>` with the appropriate `access` expression |
| Restrict an endpoint to a role | Set `access="hasRole('ROLE_MANAGER')"` on the matching intercept rule |
| Switch to LDAP authentication | Replace the `authentication-provider` with a `<security:ldap-authentication-provider>` |
| Enable HTTPS redirect | Add `<security:port-mappings>` and `requires-channel="https"` |

### dispatcher-servlet.xml

Controls the Spring MVC front controller. Key elements:

```xml
<!-- dispatcher-servlet.xml (excerpt) -->
<mvc:annotation-driven />

<!-- JSP view resolver -->
<bean class="org.springframework.web.servlet.view.InternalResourceViewResolver">
    <property name="prefix" value="/WEB-INF/views/" />
    <property name="suffix" value=".jsp" />
</bean>

<!-- Static resources served directly by the container -->
<mvc:resources mapping="/resources/**" location="/resources/" />

<!-- Internationalisation -->
<bean id="messageSource"
      class="org.springframework.context.support.ResourceBundleMessageSource">
    <property name="basenames">
        <list>
            <value>i18n/messages</value>
        </list>
    </property>
    <property name="defaultEncoding" value="UTF-8" />
</bean>

<bean id="localeResolver"
      class="org.springframework.web.servlet.i18n.SessionLocaleResolver">
    <property name="defaultLocale" value="es" />
</bean>
```

### persistence.xml

Defines the JPA persistence unit. To change the target database, update the datasource JNDI name and Hibernate dialect:

```xml
<!-- persistence.xml (excerpt) -->
<persistence-unit name="myAppPU" transaction-type="JTA">

    <class>com.example.myapp.model.Usuario</class>

    <properties>
        <!-- JNDI datasource configured in the application server -->
        <property name="hibernate.connection.datasource"
                  value="java:comp/env/jdbc/myappDS" />

        <!-- Change dialect to match your database -->
        <property name="hibernate.dialect"
                  value="org.hibernate.dialect.PostgreSQLDialect" />

        <!-- DDL strategy: validate | update | create | create-drop -->
        <property name="hibernate.hbm2ddl.auto" value="validate" />

        <property name="hibernate.show_sql"   value="false" />
        <property name="hibernate.format_sql" value="true" />
    </properties>
</persistence-unit>
```

<Warning>
  Use `hibernate.hbm2ddl.auto=validate` or `none` in production. The `create` and `create-drop` strategies will destroy existing data on startup.
</Warning>

## Internationalisation

Message bundles live under `src/main/resources/i18n/`:

```
src/main/resources/i18n/
├── messages_es.properties   ← Spanish (default)
├── messages_en.properties   ← English
└── messages.properties      ← Fallback
```

Add keys using the standard `key=value` format:

```properties
# messages_es.properties
usuario.nameRequired=El nombre es obligatorio.
usuario.emailRequired=El correo electrónico es obligatorio.

# messages_en.properties
usuario.nameRequired=Name is required.
usuario.emailRequired=E-mail address is required.
```

Reference keys in JSP views with the Spring `<spring:message>` tag:

```jsp
<spring:message code="usuario.nameRequired" />
```

## RUP Global Configuration

Initialise the RUP component library once, typically in the main layout JSP or in a shared JavaScript file loaded on every page:

```javascript
// Set application locale (matches Spring localeResolver default)
$.rup.setParam("locale", "es");

// Base path for RUP static assets (CSS, images, i18n JSON)
$.rup.setParam("appResources", "../resources");

// Application context path (used by RUP components for Ajax calls)
$.rup.setParam("contextPath", "${pageContext.request.contextPath}");
```

<Info>
  The `appResources` path must point to the folder that contains the unpacked `rup-vX.Y.Z.zip` static assets. Adjust the relative path if your WAR has a non-standard resource layout.
</Info>

## Maven Build Profiles

The generated parent `pom.xml` includes profiles for common deployment environments. Activate a profile with `-P <profile-id>` during the build.

```xml
<!-- pom.xml (profiles excerpt) -->
<profiles>

    <profile>
        <id>dev</id>
        <activation><activeByDefault>true</activeByDefault></activation>
        <properties>
            <env>dev</env>
            <log.level>DEBUG</log.level>
            <datasource.jndi>java:comp/env/jdbc/myappDevDS</datasource.jndi>
        </properties>
    </profile>

    <profile>
        <id>test</id>
        <properties>
            <env>test</env>
            <log.level>INFO</log.level>
            <datasource.jndi>java:comp/env/jdbc/myappTestDS</datasource.jndi>
        </properties>
    </profile>

    <profile>
        <id>prod</id>
        <properties>
            <env>prod</env>
            <log.level>WARN</log.level>
            <datasource.jndi>java:comp/env/jdbc/myappProdDS</datasource.jndi>
        </properties>
    </profile>

</profiles>
```

Use Maven resource filtering to inject profile properties into configuration files:

```xml
<build>
    <resources>
        <resource>
            <directory>src/main/resources</directory>
            <filtering>true</filtering>
        </resource>
    </resources>
</build>
```

Then reference them in XML configuration files as `${datasource.jndi}`.

## Logging

UDA projects use SLF4J as the logging facade, backed by Log4j. The configuration file lives at:

```
src/main/resources/log4j.xml   (or log4j2.xml for Log4j 2)
```

A minimal Log4j 2 configuration:

```xml
<?xml version="1.0" encoding="UTF-8"?>
<Configuration status="WARN">
    <Appenders>
        <Console name="Console" target="SYSTEM_OUT">
            <PatternLayout pattern="%d{HH:mm:ss.SSS} [%t] %-5level %logger{36} - %msg%n" />
        </Console>
        <RollingFile name="File"
                     fileName="logs/myapp.log"
                     filePattern="logs/myapp-%d{yyyy-MM-dd}.log.gz">
            <PatternLayout pattern="%d %-5level [%t] %logger{36} - %msg%n" />
            <TimeBasedTriggeringPolicy />
        </RollingFile>
    </Appenders>
    <Loggers>
        <Logger name="com.example.myapp" level="${log.level}" additivity="false">
            <AppenderRef ref="Console" />
            <AppenderRef ref="File" />
        </Logger>
        <Root level="WARN">
            <AppenderRef ref="Console" />
        </Root>
    </Loggers>
</Configuration>
```

<Tip>
  The `${log.level}` placeholder is resolved by Maven resource filtering at build time, so the compiled WAR automatically uses the correct level for the active profile.
</Tip>
